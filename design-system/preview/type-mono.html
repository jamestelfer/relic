func groupTurns(msgs []parser.Message) []Turn {
  var turns []Turn
  var current *Turn
  for _, m := range msgs {
    // new turn on real human input
    if m.Role == "user" && messageRole(m) == "user" {
      if current != nil { turns = append(turns, *current) }
      current = &Turn{Index: len(turns)+1, Messages: []parser.Message{m}}
    }
  }
  return turns
}
